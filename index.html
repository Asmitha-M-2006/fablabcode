// FAB-LabCode Generated: Calculator App
class Calculator {
  constructor() {
    this.display = document.querySelector('.display');
    this.history = document.querySelector('.history');
    this.current = '';
    this.operator = null;
    this.prev = null;
    this.bindKeys();
  }

  bindKeys() {
    document.addEventListener('keydown', (e) => {
      if (/[0-9.]$/.test(e.key)) this.input(e.key);
      else if (['+', '-', '*', '/'].includes(e.key))
        this.setOperator(e.key);
      else if (e.key === 'Enter') this.calculate();
      else if (e.key === 'Backspace') this.backspace();
    });
  }

  input(val) {
    if (val === '.' && this.current.includes('.')) return;
    this.current += val;
    this.update();
  }

  setOperator(op) {
    if (this.current === '') return;
    this.prev = parseFloat(this.current);
    this.operator = op;
    this.history.textContent =
      `${this.prev} ${op}`;
    this.current = '';
  }

  calculate() {
    const b = parseFloat(this.current);
    const ops = {
      '+': (a, b) => a + b,
      '-': (a, b) => a - b,
      '*': (a, b) => a * b,
      '/': (a, b) =>
        b === 0 ? 'Error' : a / b
    };
    this.current =
      String(ops[this.operator](this.prev, b));
    this.operator = null;
    this.prev = null;
    this.update();
  }

  update() {
    this.display.textContent =
      this.current || '0';
  }
}

How it works

This Calculator class encapsulates all state and logic using ES6 class syntax. The core pattern is a simple state machine with three variables: current (active input), prev (left operand), and operator (pending operation).

1
Constructor — queries DOM elements and calls bindKeys() to wire up keyboard events globally.
2
input(val) — appends digit/decimal to this.current. Guards against multiple decimal points.
3
setOperator(op) — saves current value as prev, stores the operator, and clears current.
4
calculate() — uses an ops lookup object to compute the result. Returns "Error" for division by zero instead of throwing.
